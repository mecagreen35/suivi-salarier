INSERT INTO public.pointage_demandes (
    "ID_Employe", 
    date_concernee, 
    type_demande, 
    type_absence, 
    periode_absence, 
    commentaire_salarie, 
    statut
)
SELECT 
    {{ Get_Mon_Compteur.data[0].ID_Employe }},
    d.date_jour,
    'ABSENCE',
    {{ Sel_Type_Abs.selectedOptionValue }},
    {{ Sel_Periode_Abs.selectedOptionValue }},
    {{ Inp_Comm_Abs.text || null }},
    'EN_ATTENTE'
FROM (
    SELECT generate_series(
        {{ moment(Date_Abs_Debut.selectedDate).format('YYYY-MM-DD') }}::date, 
        {{ moment(Date_Abs_Fin.selectedDate).format('YYYY-MM-DD') }}::date, 
        '1 day'::interval
    )::date AS date_jour
) d
WHERE EXTRACT(ISODOW FROM d.date_jour) <= 5;
